import numpy as np
from gitsource import GithubRepositoryDataReader, chunk_documents
from embedder import Embedder
import minsearch
def rrf(result_lists, k=60, num_results=5):
    scores = {}
    docs = {}
    for results in result_lists:
        for rank, doc in enumerate(results):
            key = (doc["filename"], doc["start"])
            scores[key] = scores.get(key, 0) + 1 / (k + rank)
            docs[key] = doc
    ranked = sorted(scores, key=scores.get, reverse=True)
    return [docs[key] for key in ranked[:num_results]]
def main():
    embedder = Embedder()
    reader = GithubRepositoryDataReader(
        repo_owner="DataTalksClub",
        repo_name="llm-zoomcamp",
        commit_id="8c1834d",
        allowed_extensions={"md"},
        filename_filter=lambda path: "/lessons/" in path,
    )
    documents = [file.parse() for file in reader.read()]
    chunks = chunk_documents(documents, size=2000, step=1000)
    text_index = minsearch.Index(text_fields=["content"], keyword_fields=[])
    text_index.fit(chunks)
    embeddings = [embedder.encode(chunk['content']) for chunk in chunks]
    X = np.array(embeddings)
    vector_index = minsearch.VectorSearch()
    vector_index.fit(X, chunks)
    query = "How do I give the model access to tools?"
    v_query = embedder.encode(query)
    text_results = text_index.search(query, num_results=10)
    vector_results = vector_index.search(v_query, num_results=10)
    hybrid_results = rrf([vector_results, text_results], k=60, num_results=5) 
    print("\n--- Top Hybrid Search Result After RRF ---")
    if hybrid_results:
        print(f"Filename: {hybrid_results[0]['filename']}")
        print(f"Start offset: {hybrid_results[0].get('start')}")
    else:
        print("No results found.")
if __name__ == "__main__":
    main()
